          setSelectedProductId(null);

                    <div key={fb.id} className="mb-4 bg-yellow-50 p-4 rounded-xl border border-yellow-200">
                      <p className="font-bold text-yellow-800 mb-2">あなたが送信したフィードバック:</p>
                      <p className="text-gray-800 bg-white p-3 rounded-lg border border-yellow-100 mb-3">{fb.text}</p>

                      {fb.reply && (
                        <div className="bg-white p-4 rounded-lg border-l-4 border-yellow-500 shadow-sm">
                          <p className="text-xs font-bold text-gray-500 mb-1">担当者 {p.authorId} からの返信:</p>
                          <p className="text-gray-800 font-bold">{fb.reply}</p>
                        </div>
                      )}
                    </div>
                  ))}

                  <div className="mt-6 flex flex-col sm:flex-row gap-2">
                    <input
                      type="text"
                      placeholder="この商品知識にアドバイスを送る..."
                      className="flex-1 p-3 border rounded-lg focus:outline-none focus:ring-2 focus:ring-yellow-400"
                      value={feedbackInput[p.id] || ''}
                      onChange={e => setFeedbackInput({...feedbackInput, [p.id]: e.target.value})}
                    />
                    <button
                      onClick={() => submitFeedback(p.id)}
                      className="bg-yellow-500 text-white px-6 py-3 rounded-lg font-bold hover:bg-yellow-600 transition-colors whitespace-nowrap"
                    >
                      FBを送信
                    </button>
                  </div>
                </div>
              )}
            </div>
          </div>
        );
      }

      return (
        <div className="max-w-4xl mx-auto space-y-6">
          <div className="bg-gradient-to-r from-[#004098] to-blue-600 rounded-2xl p-8 text-white shadow-lg flex items-center justify-between">
            <div>
              <p className="text-blue-200 font-bold mb-1">担当者番号: {currentUser}</p>
              <h2 className="text-3xl font-black">マイページ</h2>
            </div>
            <div className="text-center">
              <p className="text-sm font-bold text-blue-200">現在のレベル</p>
              <div className="text-5xl font-black text-yellow-400 drop-shadow-md">Lv.{myLevel}</div>
            </div>
          </div>

          <div className="flex border-b border-gray-200 mb-4 overflow-x-auto">
            <button
              onClick={() => handleTabChange('myProducts')}
              className={`px-6 py-3 font-bold transition-colors whitespace-nowrap flex items-center gap-2 ${activeSubTab === 'myProducts' ? 'border-b-4 border-[#004098] text-[#004098]' : 'text-gray-500 hover:text-gray-800'}`}
            >
              登録データ (FB確認・返信)
              {unrepliedFbCount > 0 && (
                <span className="bg-red-500 text-white text-xs px-2 py-0.5 rounded-full shadow-sm animate-pulse">{unrepliedFbCount}</span>
              )}
            </button>
            <button
              onClick={() => handleTabChange('requests')}
              className={`px-6 py-3 font-bold transition-colors whitespace-nowrap flex items-center gap-2 ${activeSubTab === 'requests' ? 'border-b-4 border-yellow-500 text-[#004098]' : 'text-gray-500 hover:text-gray-800'}`}
            >
              FB依頼・送信履歴
              {pendingRequestCount > 0 && (
                <span className="bg-red-500 text-white text-xs px-2 py-0.5 rounded-full shadow-sm animate-pulse">{pendingRequestCount}</span>
              )}
            </button>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            {activeSubTab === 'myProducts' && (
              <>
                {myProducts.length === 0 && <p className="text-gray-500 col-span-full py-8 text-center">まだ登録データがありません。</p>}
                {myProducts.map(p => {
                  const pUnrepliedCount = (p.feedbacks || []).filter(f => !f.reply).length;
                  return (
                    <div
                      key={p.id}
                      onClick={() => handleSelectProduct(p.id)}
                      className="bg-white p-5 rounded-xl shadow-sm border border-gray-100 cursor-pointer hover:shadow-md hover:border-blue-300 transition-all group flex justify-between items-center relative"
                    >
                      <div>
                        <span className="text-xs font-bold bg-blue-50 text-[#004098] px-2 py-1 rounded mb-2 inline-block">{p.maker}</span>
                        <h4 className="font-bold text-lg group-hover:text-[#004098] transition-colors">{p.name}</h4>
                        <p className="text-sm text-gray-500">{p.model}</p>

                        {p.feedbacks && p.feedbacks.length > 0 && (
                          <p className="text-xs font-bold text-[#004098] mt-3 bg-blue-50 inline-block px-2 py-1 rounded">
                            💬 フィードバック {p.feedbacks.length}件
                          </p>
                        )}
                        {pUnrepliedCount > 0 && (
                          <span className="text-xs font-bold text-white bg-red-500 mt-3 inline-block px-2 py-1 rounded ml-2 shadow-sm">
                            未返信 {pUnrepliedCount}件
                          </span>
                        )}
                      </div>
                      <div className="text-gray-300 group-hover:text-[#004098] group-hover:translate-x-1 transition-all">
                        <svg className="w-8 h-8" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M9 5l7 7-7 7"></path></svg>
                      </div>
                    </div>
                  );
                })}
              </>
            )}

            {activeSubTab === 'requests' && (
              <>
                {requestedProducts.length === 0 && <p className="text-gray-500 col-span-full py-8 text-center">あなた宛のフィードバック依頼や履歴はありません。</p>}
                {requestedProducts.map(p => {
                  const hasReplied = (p.feedbacks || []).some(f => f.senderId === currentUser);
                  const needsAction = p.feedbackTarget === currentUser && !hasReplied;

                  return (
                    <div
                      key={p.id}
                      onClick={() => handleSelectProduct(p.id)}
                      className="bg-white p-5 rounded-xl shadow-sm border border-yellow-200 cursor-pointer hover:shadow-md hover:border-yellow-400 transition-all group flex justify-between items-center"
                    >
                      <div>
                        <span className="text-xs font-bold bg-yellow-100 text-yellow-800 px-2 py-1 rounded mb-2 inline-block">依頼元: {p.authorId}</span>
                        <h4 className="font-bold text-lg group-hover:text-yellow-600 transition-colors">{p.name}</h4>
                        <p className="text-sm text-gray-500">{p.maker} / {p.model}</p>

                        {hasReplied ? (
                          <p className="text-xs font-bold text-yellow-600 mt-3 bg-yellow-50 inline-block px-2 py-1 rounded">
                            ✅ FB送信済み
                          </p>
                        ) : needsAction ? (
                          <p className="text-xs font-bold text-white bg-red-500 mt-3 inline-block px-2 py-1 rounded shadow-sm">
                            🚨 FB依頼が来ています！
                          </p>
                        ) : null}
                      </div>
                      <div className="text-gray-300 group-hover:text-yellow-500 group-hover:translate-x-1 transition-all">
                        <svg className="w-8 h-8" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M9 5l7 7-7 7"></path></svg>
                      </div>
                    </div>
                  );
                })}
              </>
            )}
          </div>
        </div>
      );
    };

    const RankingView = ({ products }) => {
      const userCounts = products.reduce((acc, p) => {
        acc[p.authorId] = (acc[p.authorId] || 0) + 1;
        return acc;
      }, {});

      const ranking = Object.entries(userCounts)
        .map(([id, count]) => ({ id, level: count + 1 }))
        .sort((a, b) => b.level - a.level);

      const getMedalColor = (index) => {
        if (index === 0) return 'bg-yellow-400 text-yellow-900';
        if (index === 1) return 'bg-gray-300 text-gray-800';
        if (index === 2) return 'bg-amber-600 text-white';
        return 'bg-blue-100 text-blue-800';
      };

      return (
        <div className="max-w-2xl mx-auto">
          <div className="bg-white rounded-2xl shadow-sm border border-gray-100 overflow-hidden">
            <div className="bg-[#004098] p-6 text-white text-center">
              <h2 className="text-2xl font-black tracking-wider">マスターランキング</h2>
              <p className="text-sm text-blue-200 mt-1">知識を共有してトップを目指そう！</p>
            </div>
            <div className="p-0">
              {ranking.map((user, index) => (
                <div key={user.id} className="flex items-center p-4 border-b border-gray-100 hover:bg-gray-50 transition-colors">
                  <div className={`w-10 h-10 rounded-full flex items-center justify-center font-black mr-4 ${getMedalColor(index)}`}>
                    {index + 1}
                  </div>
                  <div className="flex-1">
                    <p className="text-sm text-gray-500 font-bold">担当者番号</p>
                    <p className="text-lg font-bold text-gray-800">{user.id}</p>
                  </div>
                  <div className="text-right">
                    <p className="text-sm text-gray-500 font-bold">レベル</p>
                    <p className="text-2xl font-black text-[#004098]">Lv.{user.level}</p>
                  </div>
                </div>
              ))}
              {ranking.length === 0 && <p className="p-8 text-center text-gray-500">まだランキングデータがありません。</p>}
            </div>
          </div>
        </div>
      );
    };

    function App() {
      const [currentUser, setCurrentUser] = useState(null);
      const [activeTab, setActiveTab] = useState('search');
      const [products, setProducts] = useState(() => {
        const saved = localStorage.getItem('consulMasterData');
        return saved ? JSON.parse(saved) : [];
      });

      useEffect(() => {
        localStorage.setItem('consulMasterData', JSON.stringify(products));
      }, [products]);

      const handleAddProduct = (newProduct) => setProducts([newProduct, ...products]);

      const handleAddFeedback = (productId, text) => {
        setProducts(products.map(p => {
          if (p.id === productId) {
            const newFeedback = { id: Date.now(), senderId: currentUser, text, reply: '' };
            return { ...p, feedbacks: [...(p.feedbacks || []), newFeedback] };
          }
          return p;
        }));
      };

      const handleAddReply = (productId, feedbackId, text) => {
        setProducts(products.map(p => {
          if (p.id === productId) {
            return {
              ...p,
              feedbacks: p.feedbacks.map(f => f.id === feedbackId ? { ...f, reply: text } : f)
            };
          }
          return p;
        }));
      };

      const handleDeleteProduct = (productId) => setProducts(products.filter(p => p.id !== productId));

      const handleUpdateProduct = (updatedProduct) => {
        setProducts(products.map(p => p.id === updatedProduct.id ? updatedProduct : p));
      };

      if (!currentUser) {
        return (
          <>
            <style>{customStyles}</style>
            <LoginScreen onLogin={setCurrentUser} />
          </>
        );
      }

      const myLevel = products.filter(p => p.authorId === currentUser).length + 1;

      const unrepliedFbCount = products
        .filter(p => p.authorId === currentUser)
        .reduce((acc, p) => acc + (p.feedbacks ? p.feedbacks.filter(f => !f.reply).length : 0), 0);

      const pendingRequestCount = products
        .filter(p => p.feedbackTarget === currentUser)
        .reduce((acc, p) => {
          const hasReplied = p.feedbacks && p.feedbacks.some(f => f.senderId === currentUser);
          return acc + (hasReplied ? 0 : 1);
        }, 0);

      const totalNotifications = unrepliedFbCount + pendingRequestCount;

      return (
        <>
          <style>{customStyles}</style>
          <div className="min-h-screen bg-blue-50/50">
            <header className="bg-[#004098] text-white p-4 shadow-md sticky top-0 z-50">
              <div className="max-w-6xl mx-auto flex justify-between items-center">
                <h1 className="text-xl md:text-2xl font-black logo-text flex items-center gap-2">
                   <svg className="w-6 h-6 text-yellow-400" fill="currentColor" viewBox="0 0 24 24"><path d="M5 16L3 5l5.5 5L12 4l3.5 6L21 5l-2 11H5z" /></svg>
                   コンサルマスター
                </h1>
                <div className="flex items-center gap-4 text-sm md:text-base">
                  <span className="font-bold">担当者: {currentUser}</span>
                  <span className="bg-yellow-400 text-yellow-900 px-3 py-1 rounded-full font-black">Lv.{myLevel}</span>
                  <button onClick={() => setCurrentUser(null)} className="text-blue-200 hover:text-white text-sm underline">ログアウト</button>
                </div>
              </div>
            </header>

            <div className="bg-white border-b shadow-sm sticky top-[68px] z-40">
              <div className="max-w-6xl mx-auto flex overflow-x-auto">
                {[
                  { id: 'search', label: '商品検索' },
                  { id: 'register', label: '商品登録' },
                  { id: 'mypage', label: 'マイページ', badge: totalNotifications },
                  { id: 'ranking', label: 'ランキング' }
                ].map(tab => (
                  <button
                    key={tab.id}
                    onClick={() => setActiveTab(tab.id)}
                    className={`flex-1 py-4 px-3 sm:px-6 font-bold text-center whitespace-nowrap border-b-4 transition-colors relative flex items-center justify-center gap-2 ${
                      activeTab === tab.id ? 'border-[#004098] text-[#004098]' : 'border-transparent text-gray-500 hover:text-gray-800 hover:bg-gray-50'
                    }`}
                  >
                    {tab.label}
                    {tab.badge > 0 && (
                      <span className="bg-red-500 text-white text-[10px] font-black px-2 py-0.5 rounded-full animate-pulse shadow-sm">
                        {tab.badge}
                      </span>
                    )}
                  </button>
                ))}
              </div>
            </div>

            <main className="max-w-6xl mx-auto p-4 md:p-8">
              {activeTab === 'register' && <RegistrationForm onSubmit={handleAddProduct} currentUser={currentUser} />}
              {activeTab === 'search' && <SearchView products={products} />}
              {activeTab === 'mypage' && (
                <MyPage
                  products={products}
                  currentUser={currentUser}
                  onAddFeedback={handleAddFeedback}
                  onAddReply={handleAddReply}
                  onDeleteProduct={handleDeleteProduct}
                  onUpdateProduct={handleUpdateProduct}
                  unrepliedFbCount={unrepliedFbCount}
                  pendingRequestCount={pendingRequestCount}
                />
              )}
              {activeTab === 'ranking' && <RankingView products={products} />}
            </main>
          </div>
        </>
      );
    }

    const root = ReactDOM.createRoot(document.getElementById('root'));
    root.render(<App />);
  </script>
